func slL4NoProcesses(c slRunnerConfig, worker string) error {
	dirs, e := os.ReadDir("/proc")
	if e != nil {
		return e
	}
	for _, d := range dirs {
		if _, e = strconv.Atoi(d.Name()); e != nil {
			continue
		}
		raw, e := os.ReadFile(filepath.Join("/proc", d.Name(), "cmdline"))
		if os.IsNotExist(e) {
			continue
		}
		if e != nil {
			return fmt.Errorf("cannot establish process absence")
		}
		for _, arg := range bytes.Split(raw, []byte{0}) {
			if string(arg) == c.RootDir || string(arg) == c.JournalPath || string(arg) == worker || string(arg) == c.Server.UnixSocket || string(arg) == c.SigningKeyFile || string(arg) == filepath.Join(filepath.Dir(c.RootDir), "runner.json") {
				return fmt.Errorf("original fixture process still present: pid %s", d.Name())
			}
		}
	}
	if _, e = os.Lstat(c.Server.UnixSocket); !os.IsNotExist(e) {
		return fmt.Errorf("original UDS still present; no automatic removal")
	}
	return nil
}
func slL4Docker(ctx context.Context, c slRunnerConfig, args ...string) ([]byte, error) {
	if len(args) == 0 || args[0] != "inspect" && args[0] != "ps" {
		return nil, domain.ErrForbidden
	}
	child, end := context.WithTimeout(ctx, 5*time.Second)
	defer end()
	binary := c.DockerBinary
	if binary == "" {
		binary = "/usr/bin/docker"
	}
	cmd := exec.CommandContext(child, binary, append([]string{"--host", c.DockerHost}, args...)...)
	cmd.Env = []string{"PATH=/usr/bin:/bin", "LC_ALL=C", "XDG_RUNTIME_DIR=/run/user/1000"}
	var stderr bytes.Buffer
	cmd.Stderr = &stderr
	b, e := cmd.Output()
	if e != nil {
		return nil, fmt.Errorf("bounded read-only Docker %s failed", args[0])
	}
	if len(b) > 2<<20 {
		return nil, domain.ErrInvalid
	}
	return b, nil
}
func slL4InitialRun(r persistence.Run, want flow.State) error {
	if r.ID != slL4Run || r.TenantID != slL4Tenant || r.RunnerID != "application-fault-runner" || r.State.Status != domain.StatusRunning || r.State.Version != 9 || r.State.Lease.Epoch != 2 || r.State.Lease.Owner != "sl-L4-0" || r.State.PendingEffect == nil || r.State.PendingEffect.ID != slL4Op || r.State.PendingEffect.ArgsHash != "ef77c8bf901c5eb227484966ea93b183018290a713d4f2b3dc8d41e77f090af1" {
		return fmt.Errorf("original run binding differs")
	}
	x, e := domain.CanonicalJSON(slJSON(r.State))
	if e != nil {
		return e
	}
	y, e := domain.CanonicalJSON(slJSON(want))
	if e != nil || !bytes.Equal(x, y) {
		return fmt.Errorf("original snapshot changed since retained observation")
	}
	return nil
}
func slL4Terminal(r, before persistence.Run) error {
	if r.ID != before.ID || r.TenantID != before.TenantID || r.State.Status != domain.StatusCancelled || r.State.PendingEffect == nil || r.State.PendingEffect.ID != slL4Op || r.State.PendingEffect.Status != flow.EffectFailed || r.State.PendingEffect.ReceiptRef == "" || r.State.PendingEffect.DispatchEpoch != 2 || len(r.State.RemainingEffects) != 0 || r.State.Lease.Epoch <= 2 || r.State.Verification == domain.VerificationVerified || !r.State.Limits.Deadline.Equal(before.State.Limits.Deadline) || r.State.Cost != before.State.Cost || r.State.ModelRounds != before.State.ModelRounds || r.State.ToolCalls != before.State.ToolCalls || r.State.WorkspaceRevision != before.State.WorkspaceRevision {
		return fmt.Errorf("cancel result changed execution/budget/verification contract")
	}
	return nil
}

		if slSHA(b) != ref.SHA256 || int64(len(b)) != ref.ByteSize {
			t.Fatal("READY artifact bytes differ")
		}
		check(slWrite(filepath.Join(out, "artifacts", string(ref.ID)+".bytes"), b))
		if ref.ObjectKey == job.Log.Artifact.ObjectKey {
			if ref.Kind != "operation_log" {
				t.Fatal("log kind differs")
			}
			logCount++
		}
		if string(ref.ID) == cleanup.SnapshotRef {
			if ref.Kind != "workspace_snapshot" {
				t.Fatal("snapshot kind differs")
			}
			check(slL4Snapshot(b, cleanup.RunnerEpoch))
			snapshotRef = artifact.Ref{TenantID: ref.TenantID, RunID: ref.RunID, Kind: ref.Kind, ObjectKey: ref.ObjectKey, SHA256: ref.SHA256, Size: ref.ByteSize}
			snapshotCount++
		}
	}
	if logCount != 1 || snapshotCount != 1 {
		t.Fatal("single READY original log/code snapshot missing")
	}
	pgAfter, e := slL4PG(ctx, db)
	check(e)
	check(slL4PGUnchanged(pgBefore, pgAfter))
	f.save("postgres-after.json", pgAfter)
	check(db.Pool.QueryRow(ctx, `SELECT (SELECT coalesce(sum(active_count),0) FROM tenant_runtime),(SELECT reserved_slots FROM runners WHERE id='application-fault-runner'),(SELECT count(*) FROM runner_allocations WHERE state<>'released'),(SELECT count(*) FROM model_attempts),(SELECT count(*) FROM effects),(SELECT count(*) FROM workspace_cleanup WHERE phase='released')`).Scan(&active, &reserved, &allocations, &attempts, &effects, &cleanups))
	if active != 0 || reserved != 0 || allocations != 0 || attempts != 4 || effects != 8 || cleanups != 1 {
		t.Fatal("capacity/fee/effect closure differs")
	}
	var joined, confirmations, unsettled int
	check(db.Pool.QueryRow(ctx, `SELECT (SELECT count(*) FROM effects e JOIN artifacts a ON a.tenant_id=e.tenant_id AND a.id=e.receipt_ref WHERE e.tenant_id=$1 AND e.run_id=$2 AND e.operation_id=$3 AND e.epoch=2 AND e.status='failed' AND a.kind='operation_receipt' AND a.state='ready'),(SELECT count(*) FROM run_snapshots WHERE tenant_id=$1 AND run_id=$2 AND input_event->>'kind'='reconciled' AND input_event->'receipt'->>'effect_id'=$3),(SELECT count(*) FROM effects WHERE status NOT IN ('planned','skipped','succeeded','failed','cancelled'))`, slL4Tenant, slL4Run, slL4Op).Scan(&joined, &confirmations, &unsettled))
	if joined != 1 || confirmations != 1 || unsettled != 0 {
		t.Fatal("single original effect/receipt reconciliation missing")
	}
	f.save("closure-oracle.json", map[string]any{"tenant_active": active, "runner_reserved": reserved, "unreleased_allocations": allocations, "model_attempts": attempts, "effects": effects, "released_cleanup": cleanups, "joined_original_epoch2_failed_receipt": joined, "durable_reconciled_confirmations": confirmations, "unsettled_effects": unsettled, "synthetic_committed_tokens_before": committedTokens, "synthetic_committed_microusd_before": committedMoney, "protected_model_attempts_and_quotas_unchanged": true})
	f.save("control-rpc-calls.json", guard.Calls)
	f.stopRunner()
	final, e := slJournalSnapshot(ctx, c.JournalPath)
	check(e)
	check(slIdle(final))
	check(slL4ProtectedJournal(j, final))
	check(slReservation(final, slL4Tenant, slL4Run, c.Logs, 2))
	check(slVerifyPins(final, receipt, job.Log.Artifact, snapshotRef))
	removed := 0
	for _, lr := range final.Tables["operation_logs"] {
		if lr["operation_id"] == slL4Op || lr["operation_id"] == slL4Run+"_0_initial_target" {
			if lr["cleanup_state"] != "removed" {
				t.Fatal("owned container removal unconfirmed")
			}
			ids, e := slL4Docker(ctx, c, "ps", "--all", "--no-trunc", "--filter", "id="+fmt.Sprint(lr["container_id"]), "--format", "{{.ID}}")
			check(e)
			if strings.TrimSpace(string(ids)) != "" {
				t.Fatal("original container remains")
			}
			removed++
		}
	}
	if removed != 2 {
		t.Fatal("both original process-operation containers require removal proof")
	}
	if _, e = os.Lstat(workspace); !os.IsNotExist(e) {
		t.Fatal("released workspace directory remains")
	}
	f.save("final-journal.json", final)
	phase = "released"
	passed = true
}
