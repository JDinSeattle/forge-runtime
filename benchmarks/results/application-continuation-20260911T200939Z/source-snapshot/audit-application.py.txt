#!/usr/bin/python3
"""Independently cross-check retained application-fault evidence; no services or DB writes."""
import argparse
import base64
import datetime
import hashlib
import json
from pathlib import Path


def read(path):
    return json.loads(path.read_text())


def instant(value):
    return datetime.datetime.fromisoformat(value)


def nanos(value):
    elapsed = instant(value) - datetime.datetime(1970, 1, 1, tzinfo=datetime.timezone.utc)
    return (elapsed.days * 86400 + elapsed.seconds) * 1_000_000_000 + elapsed.microseconds * 1000


def receipt_matches(operation, archived):
    # The artifact has already been size/digest checked against PG. Compare its
    # payload, not only the receipt ref repeated in two mutable summary files.
    for field in ('request', 'status', 'job_id', 'before_hash', 'after_hash', 'after_revision', 'result', 'cancel_requested'):
        assert archived.get(field) == operation.get(field)


def effect_matches(effect, request):
    for field in ('tenant_id', 'run_id', 'operation_id', 'epoch', 'kind', 'args', 'args_hash', 'policy_version', 'expected_revision'):
        assert effect[field] == request[field]
    canonical = bytes.fromhex(effect['canonical_args'].removeprefix('\\x'))
    assert json.loads(canonical) == request['args']
    assert hashlib.sha256(canonical).hexdigest() == request['args_hash']


def audit(directory, mode):
    proof = read(directory / 'acceptance.json')
    final = read(directory / 'cleanup-released-postgres.json')
    before = read(directory / 'before-worker-death-postgres.json')
    dead = read(directory / 'after-worker1-sigkill-postgres.json')
    operation = read(directory / 'original-operation.json')
    role = read(directory / 'worker-role-preflight.json')
    run = final['runs'][0]
    run_id, tenant = run['id'], run['tenant_id']
    assert len(final['runs']) == 1
    for phase in (before, dead):
        assert len(phase['runs']) == 1
        for field in ('id', 'task', 'tenant_id', 'project_id', 'principal_id', 'parent_run_id', 'base_commit', 'input_snapshot', 'config_snapshot', 'created_at', 'runner_id', 'workspace_id'):
            assert phase['runs'][0][field] == run[field]
    assert run['state'] == run['snapshot']['status'] == proof['terminal_status'] == 'completed'
    assert run['snapshot']['verification_status'] == proof['verification'] == 'verified'
    assert role['superuser'] is False and role['explicit_bypassrls'] is True
    assert role['current_user'] == role['session_user']
    assert all(role[key] is False for key in ('create_role', 'create_database', 'create_schema', 'member_of_runs_owner'))
    assert proof['worker1_pid'] != proof['worker2_pid']
    if mode != 'F09':
        assert all(before[table][0][field] == 1 for table, field in (('tenant_runtime', 'active_count'), ('runners', 'reserved_slots')))
    lease = dead['runs'][0]
    assert lease['lease_epoch'] == proof['original_epoch']
    if mode != 'F09':
        assert instant(lease['lease_until']) == instant(proof['original_lease_until'])
    claims = [row['input_event'] for row in final['run_snapshots'] if (row['input_event'] or {}).get('kind') == 'claimed' and row['input_event']['epoch'] > proof['original_epoch']]
    claim = min(claims, key=lambda event: event['epoch'])
    assert instant(claim['at']) == instant(proof['replacement_claim_db_time'])
    assert claim['epoch'] == proof['original_epoch'] + 1
    assert claim['owner'] == claim['lease']['owner'] == 'worker2-0'
    assert claim['lease']['epoch'] == claim['epoch']
    if mode != 'F09':
        assert instant(claim['at']) >= instant(lease['lease_until']) > instant(proof['worker1_sigkill_at'])
    attempts = final['model_attempts']
    reservations = final['quota_reservations']
    assert len(attempts) == len(reservations) == 4
    assert {row['attempt_id'] for row in attempts} == {row['id'] for row in reservations}
    assert all(row['provider'] == 'fake' and row['status'] == 'completed' for row in attempts)
    assert all(row['status'] == 'settled' and row['request_slot_released'] for row in reservations)
    cost = sum(row['actual_microusd'] for row in reservations)
    tokens = sum(row['actual_tokens'] for row in reservations)
    quota = final['provider_quotas'][0]
    assert cost == tokens == quota['committed_microusd'] == quota['committed_tokens'] == run['snapshot']['cost_microusd'] == 570
    assert all(quota[key] == 0 for key in ('active_requests', 'reserved_microusd', 'reserved_tokens'))
    assert final['tenant_runtime'][0]['active_count'] == final['runners'][0]['reserved_slots'] == 0
    target = [row for row in final['effects'] if row['operation_id'] == proof['target_operation']]
    assert len(target) == 1
    target = target[0]
    effect_matches(target, operation['request'])
    assert target['epoch'] == operation['request']['epoch'] == proof['target_dispatch_epoch']
    assert target['args_hash'] == operation['request']['args_hash']
    assert target['status'] == operation['status'] == ('cancelled' if mode == 'F08' else 'succeeded')
    artifacts = {row['id']: row for row in final['artifacts']}
    assert artifacts[target['receipt_ref']]['sha256'] == operation['receipt']['sha256']
    for identity, metadata in artifacts.items():
        raw = (directory / 'artifacts' / (identity + '.json')).read_bytes()
        assert metadata['state'] == 'ready' and metadata['tenant_id'] == tenant and metadata['run_id'] == run_id
        assert hashlib.sha256(raw).hexdigest() == metadata['sha256'] and len(raw) == metadata['byte_size']
    stored_operation = read(directory / 'artifacts' / (target['receipt_ref'] + '.json'))
    receipt_matches(operation, stored_operation)
    stdout = (directory / 'original-command.stdout').read_bytes()
    assert stdout == base64.b64decode(operation['result']['output'], validate=True)
    verification = read(directory / 'artifacts' / (run['snapshot']['verification_report_ref'] + '.json'))
    assert all(verification['evidence'][key] is True for key in ('trusted', 'baseline_target_failed', 'target_passed', 'regression_passed'))
    assert verification['evidence']['workspace_revision'] == run['snapshot']['workspace_revision']
    assert all(identity in artifacts for identity in verification['receipts'])
    cleanup = final['workspace_cleanup'][0]
    assert cleanup['phase'] == proof['cleanup']['phase'] == 'released'
    snapshot = artifacts[cleanup['snapshot_ref']]
    assert instant(snapshot['created_at']) <= instant(cleanup['released_at'])
    events = sorted(row['seq'] for row in final['run_events'])
    assert events == list(range(1, run['next_event_seq']))
    docker = [json.loads(line) for line in (directory / 'original-docker-events.jsonl').read_text().splitlines()]
    assert all(event['Actor']['Attributes']['name'] == operation['job_id'] for event in docker)
    assert sum(event['Action'] == 'create' for event in docker) == sum(event['Action'] == 'start' for event in docker) == 1
    extra = {}
    if mode == 'F02':
        assert all(count == 0 for count in proof['zero_external_work'].values())
        assert all(before[table] == [] for table in ('approvals', 'artifacts', 'effects', 'model_attempts', 'quota_reservations'))
        assert before['runs'][0]['snapshot']['workspace_revision'] == 0
        assert before['runs'][0]['snapshot']['step_seq'] == 0
        assert target['epoch'] > proof['original_epoch']
        extra['zero_pg_external_work_before_death'] = True
    if mode == 'F06':
        patch_before = read(directory / 'patch-before-death.json')
        patch_after = read(directory / 'patch-after-recovery.json')
        assert patch_before == patch_after
        assert patch_before['receipt'] == read(directory / 'patch-journal-receipt-before-death.json')
        patch_id = patch_before['request']['operation_id']
        old_effect = [row for row in before['effects'] if row['operation_id'] == patch_id]
        new_effect = [row for row in final['effects'] if row['operation_id'] == patch_id]
        assert len(old_effect) == len(new_effect) == 1
        assert old_effect[0]['status'] == 'in_flight' and not old_effect[0]['receipt_ref']
        assert new_effect[0]['status'] == 'succeeded'
        assert artifacts[new_effect[0]['receipt_ref']]['sha256'] == patch_after['receipt']['sha256']
        patch_archived = read(directory / 'artifacts' / (new_effect[0]['receipt_ref'] + '.json'))
        receipt_matches(patch_before, patch_archived)
        effect_matches(old_effect[0], patch_before['request'])
        effect_matches(new_effect[0], patch_after['request'])
        assert patch_before['after_revision'] == patch_before['request']['expected_revision'] + 1
        assert patch_before['before_hash'] != patch_before['after_hash'] == patch_before['expected_after_hash']
        assert old_effect[0]['epoch'] == new_effect[0]['epoch'] == patch_before['request']['epoch']
        expected = Path(__file__).resolve().parents[2] / 'testdata/repairs/clamp/expected/app.py'
        assert (directory / 'patch-before-death-app.py').read_bytes() == expected.read_bytes()
        extra['patch_receipt_and_revision_preserved'] = True
    if mode == 'F09':
        restarted = read(directory / 'approval-after-restart-postgres.json')
        assert before['runs'][0]['state'] == restarted['runs'][0]['state'] == 'waiting_approval'
        assert before['runs'][0]['version'] == restarted['runs'][0]['version']
        assert before['approvals'] == restarted['approvals']
        assert before['runs'][0]['snapshot']['approval'] == restarted['runs'][0]['snapshot']['approval']
        assert proof['runner1_pid'] != proof['runner2_pid']
        assert proof['approval_workspace_hash_before'] == proof['approval_workspace_hash_after']
        preserved = read(directory / 'approval-after-restart.json')
        assert preserved == proof['approval_preserved']
        binding = preserved['binding']
        pending = before['approvals']
        assert len(pending) == 1
        pending = pending[0]
        assert binding == before['runs'][0]['snapshot']['approval']
        assert binding['granted'] is False
        assert preserved['run_version_before'] == before['runs'][0]['version']
        assert preserved['workspace_revision'] == binding['workspace_revision'] == before['runs'][0]['snapshot']['workspace_revision']
        approved = [row for row in final['approvals'] if row['id'] == pending['id']]
        assert len(approved) == 1 and approved[0]['decision'] == 'allow'
        approved = approved[0]
        for field in ('effect_id', 'args_hash', 'workspace_revision', 'policy_version', 'version'):
            assert pending[field] == approved[field] == binding[field]
        assert pending['decision'] is None and pending['decided_at'] is None
        request = operation['request']
        assert binding['effect_id'] == request['operation_id']
        assert binding['workspace_revision'] == request['expected_revision']
        assert binding['args_hash'] == request['args_hash']
        assert binding['policy_version'] == request['policy_version']
        assert proof['approval_workspace_hash_before'] == operation['before_hash']
        decisions = [row['input_event'] for row in final['run_snapshots'] if (row['input_event'] or {}).get('kind') == 'approval_decided' and row['input_event']['approval']['binding']['effect_id'] == binding['effect_id']]
        assert len(decisions) == 1
        decision = decisions[0]
        assert decision['approval'] == {'approve': True, 'binding': binding}
        assert decision['expected_version'] == preserved['run_version_before']
        assert instant(decision['at']) == instant(approved['decided_at'])
        worker_ready = read(directory / 'worker2-ready.json')
        assert worker_ready['pid'] == proof['worker2_pid']
        assert instant(proof['runner_sigkill_at']) < instant(worker_ready['at']) < instant(approved['decided_at']) <= instant(claim['at'])
        assert all(nanos(approved['decided_at']) < event['timeNano'] for event in docker if event['Action'] in ('create', 'start'))
        assert preserved['operation_count_before_approval'] == 0
        assert all(preserved[name + '_rejected'] is True for name in ('approval_version', 'args_hash', 'effect_id', 'policy_version', 'workspace_revision'))
        extra['approval_binding_preserved_across_restart'] = True
    if mode == 'F07':
        assert before['runs'][0]['state'] == 'needs_reconciliation'
        assert any(row['operation_id'] == proof['target_operation'] and row['status'] == 'unknown' for row in before['effects'])
    if mode == 'F08':
        assert operation['result']['truncated'] is False
        times = [json.loads(line)['old_write_ns'] for line in stdout.splitlines()]
        assert times == proof['old_writer_ns']
        assert times and all(a < b for a, b in zip(times, times[1:]))
        assert times[-1] > nanos(lease['lease_until'])
        next_effect = [row for row in final['effects'] if row['operation_id'] == run_id + '_step_2_op_0']
        assert len(next_effect) == 1 and next_effect[0]['status'] == 'succeeded'
        next_receipt = read(directory / 'artifacts' / (next_effect[0]['receipt_ref'] + '.json'))
        assert next_receipt['result']['truncated'] is False
        next_stdout = base64.b64decode(next_receipt['result']['output'], validate=True)
        next_time = json.loads(next_stdout)['new_write_ns']
        assert next_time == proof['new_writer_first_ns']
        deaths = [event['timeNano'] for event in docker if event['Action'] == 'die']
        assert len(deaths) == 1 and times[-1] < deaths[0] < next_time
        extra = {'old_writer_samples': len(times), 'writer_gap_ns': next_time - times[-1], 'old_write_after_expiry_ns': times[-1] - nanos(lease['lease_until']), 'old_container_die_ns': deaths[0], 'new_write_after_old_die_ns': next_time - deaths[0]}
    else:
        assert proof['counter'] == '1\n'
    return {'case': mode, 'run_id': run_id, 'ready_artifacts_checked': len(artifacts), 'events_after_cleanup': len(events), 'synthetic_cost_microusd': cost, 'claim_after_expiry_seconds': (instant(claim['at']) - instant(lease['lease_until'])).total_seconds() if mode != 'F09' else None, **extra}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('directory', type=Path)
    parser.add_argument('--continuation', action='store_true')
    args = parser.parse_args()
    modes = ('F02', 'F06', 'F09') if args.continuation else ('F05', 'F07', 'F08')
    results = [audit(args.directory / mode, mode) for mode in modes]
    print(json.dumps({'passed': True, 'scope': 'Independent recomputation from retained fixture records; not a second execution or independent OS signal observer. Phase exports are sequential table reads, not atomic database snapshots. Model costs are synthetic.', 'cases': results}, indent=2))


if __name__ == '__main__':
    main()
