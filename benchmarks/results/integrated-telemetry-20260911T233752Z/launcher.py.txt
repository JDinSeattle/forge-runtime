env={k:v for k,v in os.environ.items() if not k.startswith(('FORGE_','OTEL_'))}

 k,sep,v=line.partition('=')
 if sep and k in ('FORGE_REVIEW_DATABASE_URL','FORGE_REVIEW_ALLOW_FIXTURES'):env[k]=shlex.split(v)[0]

 (out/'launcher.py.txt').write_text(pathlib.Path(__file__).read_text())

 source=json.loads((out/'source-inputs.json').read_text()) if (out/'source-inputs.json').exists() else {}

 (out/'manifest.json').write_text(json.dumps({str(p.relative_to(out)):hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(out.rglob('*')) if p.is_file()},indent=2)+'\n')
