from pathlib import Path
import datetime, hashlib, importlib.util, json, os, shlex, subprocess, sys, time

root = Path('/home/postedism/Desktop/mini claude code/forge-runtime')
repo = root / 'var/local/clean-clone-e49'
output = root / 'benchmarks/results/clean-clone-e49-20260912'
binaries = root / 'var/local/clean-clone-e49-binaries'
output.mkdir(mode=0o700)
binaries.mkdir(mode=0o700)
env = {k: v for k, v in os.environ.items() if not k.startswith(('FORGE_', 'OTEL_'))}
for line in (root / 'var/local/review-database.env').read_text().splitlines():
    key, sep, value = line.partition('=')
    if sep and key in ('FORGE_REVIEW_DATABASE_URL', 'FORGE_REVIEW_ALLOW_FIXTURES'):
        env[key] = shlex.split(value)[0]
assert env.get('FORGE_REVIEW_ALLOW_FIXTURES') == '1'
env.update(FORGE_TEST_DATABASE_URL=env['FORGE_REVIEW_DATABASE_URL'], GOCACHE=str(root / 'var/local/go-build-cache'), GOPROXY='off', GOSUMDB='off', GOENV='off', GOTOOLCHAIN='local', GOTMPDIR=str(root / 'var/local/integration-go-tmp'), TMPDIR='/tmp', FORGE_REVIEW_FORGE_BINARY=str(binaries / 'forge'), FORGE_REVIEW_ADMIN_BINARY=str(binaries / 'forge-admin'), FORGE_REVIEW_API_BINARY=str(binaries / 'forge-api'))

def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def dump(name, value):
    (output / name).write_text(json.dumps(value, indent=2, sort_keys=True) + '\n')

def status():
    return subprocess.check_output(['git', 'status', '--porcelain'], cwd=repo, text=True)

def snapshot():
    names = subprocess.check_output(['rg', '--files', '--hidden', '-g', '!.git', '-g', '!var', '-g', '!bin', '-g', '!benchmarks/results'], cwd=repo, text=True).splitlines()
    return {n: sha(repo / n) for n in sorted(names) if Path(n).suffix in ('.go', '.mod', '.sum', '.sql', '.yaml', '.yml', '.proto', '.py', '.sh') or n == 'Makefile'}

assert status() == ''
before = snapshot()
dump('source-before.json', before)
dump('identity.json', {'commit': subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=repo, text=True).strip(), 'clean_before': True, 'clone': str(repo), 'go': subprocess.check_output(['go', 'version'], env=env, text=True).strip(), 'scope': 'independent local clean clone with existing Go caches and private fixture PostgreSQL; actual four no-mount Docker capture cases; no services/shared volumes changed; no fresh-machine or full disk-spool claim'})
(output / 'launcher.py.txt').write_bytes(Path(__file__).read_bytes())
spec = importlib.util.spec_from_file_location('bounded', repo / 'tests/review/run_api_lifecycle.py')
bounded = importlib.util.module_from_spec(spec)
spec.loader.exec_module(bounded)
checks = [
    ('build', ['go', 'build', '-o', str(binaries) + '/', './cmd/...']),
    ('capture-build', ['go', 'test', '-race', '-c', '-o', str(binaries / 'sandbox.test'), './internal/sandbox']),
    ('capture', [str(binaries / 'sandbox.test'), '-test.run=^TestNoMountCapturedDockerAcceptance$', '-test.v', '-test.count=1', '-test.timeout=90s']),
    ('test', ['go', 'test', '-json', '-count=1', './...']),
    ('generated', ['make', 'check-generated']),
]
results = []
for name, argv in checks:
    command_env = dict(env)
    if name == 'capture':
        command_env.update(FORGE_E44_NOMOUNT_DOCKER_HOST='unix:///run/user/1000/forge-runtime-docker.sock', FORGE_E44_NOMOUNT_OUTPUT=str(output / 'capture'))
    if name == 'test':
        (output / 'api-lifecycle').mkdir()
        command_env.update(FORGE_RECOVERY_UPGRADE_OUTPUT=str(output / 'upgrade'), FORGE_HANDOFF_EVIDENCE_DIR=str(output / 'handoff'), FORGE_REVIEW_LIFECYCLE_EVIDENCE=str(output / 'api-lifecycle'))
    start = time.monotonic()
    with (output / (name + '.log')).open('xb') as log:
        code = bounded.execute_bounded(argv, repo, command_env, log, 100 if name == 'capture' else 600)
    row = {'name': name, 'argv': argv, 'exit_code': code, 'elapsed_seconds': time.monotonic() - start}
    if name == 'test':
        records = []
        for line in (output / 'test.log').read_text().splitlines():
            try:
                records.append(json.loads(line))
            except ValueError:
                pass
        row['tests'] = {action: sum(bool(r.get('Test')) and r.get('Action') == action for r in records) for action in ('pass', 'fail', 'skip')}
    results.append(row)
    dump('results.json', results)
    print(json.dumps(row), flush=True)
    if code:
        break
    if name in ('build', 'capture-build'):
        dump('binaries.json', {p.name: sha(p) for p in sorted(binaries.iterdir()) if p.is_file()})
        for p in sorted(binaries.iterdir()):
            if p.is_file():
                with (output / (p.name + '-build-info.log')).open('wb') as log:
                    subprocess.run(['go', 'version', '-m', str(p)], env=env, stdout=log, stderr=subprocess.STDOUT, check=True)
after = snapshot()
dump('source-after.json', after)
dump('final-state.json', {'source_unchanged': before == after, 'source_inputs': len(before), 'clean_after': status() == '', 'status': status(), 'all_passed': len(results) == len(checks) and all(x['exit_code'] == 0 for x in results)})
dump('manifest.json', {str(p.relative_to(output)): sha(p) for p in sorted(output.rglob('*')) if p.is_file() and p.name != 'manifest.json'})
print(json.dumps({'output': str(output), 'source_unchanged': before == after, 'clean_after': status() == ''}), flush=True)
sys.exit(0 if before == after and status() == '' and len(results) == len(checks) and all(x['exit_code'] == 0 for x in results) else 1)
