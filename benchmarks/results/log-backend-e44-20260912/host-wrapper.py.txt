"""

def run(argv,timeout=15,**kw):return subprocess.run(argv,check=True,timeout=timeout,**kw)
def text(argv):return run(argv,capture_output=True,text=True).stdout.strip()
def sha(p):return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()
def write(p,data):p.write_text(json.dumps(data,indent=2,sort_keys=True)+'\n')
def show(unit):
 fields=['ActiveState','SubState','MainPID','ControlGroup','EnvironmentFiles','ExecStart','WorkingDirectory','Delegate','Environment']
 raw=text(['systemctl','--user','show',unit,'--property='+','.join(fields)])
 return dict(s.split('=',1) for s in raw.splitlines() if '=' in s)
def sqlite_state(c):
 db=sqlite3.connect(pathlib.Path(c['journal_path']).as_uri()+'?mode=ro',uri=True);db.execute('BEGIN')
 try:
  version=db.execute('PRAGMA user_version').fetchone()[0]
  identity=db.execute('SELECT id FROM journal_identity').fetchone()[0]
  pending=db.execute("SELECT count(*) FROM operations WHERE status IN ('prepared','running','unknown')").fetchone()[0]
  leased=db.execute('SELECT count(*) FROM volume_leases WHERE released=0').fetchone()[0]
  names=[r[0] for r in db.execute("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name")]
  tables={}
  for name in names:
   assert name.replace('_','').isalnum()
   rows=db.execute('SELECT * FROM "'+name+'" ORDER BY rowid').fetchall()
   tables[name]={'rows':len(rows),'sha256':hashlib.sha256(json.dumps(rows,separators=(',',':'),ensure_ascii=True).encode()).hexdigest()}
  return {'version':version,'identity':identity,'pending_operations':pending,'unreleased_leases':leased,'tables':tables}
 finally:db.rollback();db.close()

def active(unit):return show(unit).get('ActiveState')=='active'
def validate_sources(states,c):
 expected=[('forge-api','platform.json',None),('forge-worker','platform.json','recovery-worker-1'),('forge-worker','platform.json','recovery-worker-2'),('forge-runner','runner.json',None)]
 files={str(CONFIG):sha(CONFIG),str(REPO/'var/local/platform.json'):sha(REPO/'var/local/platform.json')}
 runtime={}
 def process_identity(pid):
  path=pathlib.Path('/proc')/pid
  return {'argv':(path/'cmdline').read_bytes().rstrip(b'\0').decode().split('\0'),'cwd':str((path/'cwd').resolve()),'uid_map':(path/'uid_map').read_text(),'gid_map':(path/'gid_map').read_text(),'exe_sha256':sha(path/'exe')}
 for unit,(exe,conf,worker) in zip(UNITS,expected):
  s=states[unit];assert s['ActiveState']=='active' and s['SubState']=='running',(unit,'not running')
  binary=REPO/'bin'/exe;files[str(binary)]=sha(binary)
  expected_cwd=str(REPO) if exe!='forge-runner' else '/home/postedism'
  assert s['WorkingDirectory'].removeprefix('!')==expected_cwd
  identity=process_identity(s['MainPID']);assert identity['cwd']==expected_cwd
  expected_args=[str(binary),'-config',str(REPO/'var/local'/conf)]
  if worker:expected_args+=['-id',worker]
  if exe=='forge-runner':expected_args=['/usr/bin/rootlesskit','--propagation=rslave','--state-dir=/run/user/1000/forge-runtime-runnerkit']+expected_args
  assert identity['argv']==expected_args,(unit,'unexpected process arguments')
  assert s['Delegate']==('yes' if exe=='forge-runner' else 'no')
  if exe=='forge-runner':assert not s.get('EnvironmentFiles')
  runtime[unit]={'main':identity,'service':{'working_directory':s['WorkingDirectory'].removeprefix('!'),'environment_files':s.get('EnvironmentFiles',''),'environment':s.get('Environment',''),'delegate':s['Delegate']}}
  assert str(binary) in s['ExecStart'] and str(REPO/'var/local'/conf) in s['ExecStart']
  if worker:assert ' -id '+worker+' ' in s['ExecStart']
  if exe!='forge-runner':
   assert sha('/proc/'+s['MainPID']+'/exe')==files[str(binary)],(unit,'different running binary')
   filename=REPO/'var/local'/('api.env' if exe=='forge-api' else 'worker.env')
   assert s['EnvironmentFiles']==str(filename)+' (ignore_errors=no)'
   files[str(filename)]=sha(filename)
  else:
   assert identity['exe_sha256']==sha('/usr/bin/rootlesskit'),'running rootlesskit differs from restorable file'
   assert s['Delegate']=='yes' and '/usr/bin/rootlesskit --propagation=rslave --state-dir=/run/user/1000/forge-runtime-runnerkit ' in s['ExecStart']
   cgroup=pathlib.Path('/sys/fs/cgroup')/s['ControlGroup'].lstrip('/')
   candidates=[]
   for f in cgroup.rglob('cgroup.procs'):
    for pid in f.read_text().split():
     path=pathlib.Path('/proc')/pid/'exe'
     try:
      if path.resolve()==binary:candidates.append(pid)
     except (OSError,RuntimeError):pass
   assert len(set(candidates))==1,('runner PID identity',candidates)
   child=process_identity(candidates[0])
   assert child['exe_sha256']==files[str(binary)] and child['cwd']==expected_cwd and child['argv']==[str(binary),'-config',str(CONFIG)]
   runtime[unit]['runner_child']=child
  for entry in shlex.split(s.get('Environment','')):
   key,sep,value=entry.partition('=');assert sep and unit==UNITS[2] and key=='FORGE_METRICS_LISTEN'
   host,port=value.rsplit(':',1);assert host=='127.0.0.1' and 1024<=int(port)<=65535
 return files,runtime

def test_unit_state(unit):
 p=subprocess.run(['systemctl','--user','show',unit,'--property=LoadState,ActiveState,SubState,MainPID,ControlGroup'],capture_output=True,text=True,timeout=10)
 state=dict(s.split('=',1) for s in p.stdout.splitlines() if '=' in s)
 if p.returncode and state.get('LoadState')!='not-found':raise RuntimeError('could not inspect exact test unit')
 return state

def stop_test_group(unit,known_state=None):
 before=test_unit_state(unit)
 stopped=subprocess.run(['systemctl','--user','stop',unit],capture_output=True,timeout=25)
 after=test_unit_state(unit)
 assert after.get('LoadState')=='not-found' or (after.get('ActiveState') in ('inactive','failed') and after.get('MainPID')=='0'), 'test unit termination not confirmed'
 for state in [known_state or {},before,after]:
  group=state.get('ControlGroup')
  if not group:continue
  assert group.startswith('/') and '..' not in pathlib.PurePosixPath(group).parts
  path=pathlib.Path('/sys/fs/cgroup')/group.lstrip('/')
  if path.exists():
   for members in path.rglob('cgroup.procs'):assert not members.read_text().strip(), 'test cgroup still has executing PIDs'
 return {'stop_exit_code':stopped.returncode,'before':before,'after':after,'no_executing_test_pids':True}
def cleanup_fixture(cfg,out):
 actions=[]
 backend=out/'backend'
 if not backend.exists():return actions
 for path in sorted(backend.glob('forge-e44-*-intent.json')):
  plan=json.loads(path.read_text());op=plan['operation'];name=op['operation_id']
  assert re.fullmatch(r'forge-e44-[0-9a-f]{16}-(binary|flood|cancel|shutdown)',name)
  assert path.name==name+'-intent.json' and plan['expected_labels']=={'forge.runtime':'1','forge.operation_id':name}
  cmd=['docker','--host',cfg['docker_host']]
  found=subprocess.run(cmd+['container','inspect',name],capture_output=True,text=True,timeout=10)
  if found.returncode:
   assert 'No such container' in found.stderr or 'No such object' in found.stderr, 'unconfirmed fixture lookup'
   actions.append({'name':name,'absent':True});continue
  rows=json.loads(found.stdout);assert len(rows)==1;facts=rows[0];identity=facts['Id']
  assert re.fullmatch('[0-9a-f]{64}',identity) and facts['Name']=='/'+name
  if plan.get('container_id'):assert identity==plan['container_id']
  labels=facts['Config']['Labels'];assert labels['forge.runtime']=='1' and labels['forge.operation_id']==name
  assert facts['Config']['Image']==cfg['image'] and facts['HostConfig']['LogConfig']['Type']=='none'
  if facts['State']['Running']:run(cmd+['container','kill','--signal=KILL',identity],capture_output=True,text=True,timeout=15)
  # Never use force-remove. Remove only the observed stopped, identity-checked fixture.
  current=json.loads(text(cmd+['container','inspect',identity]))[0]
  assert current['Id']==identity and current['Name']=='/'+name and not current['State']['Running']
  run(cmd+['container','rm',identity],capture_output=True,text=True,timeout=15)
  actions.append({'name':name,'container_id':identity,'fixture_fallback_removal':True})
 return actions
