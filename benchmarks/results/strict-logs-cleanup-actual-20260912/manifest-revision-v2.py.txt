"""Revise only the never-executed logs02 binary binding; retain old preparation."""
import fcntl
import importlib.util
import json
import os
from pathlib import Path
import re
import sqlite3
import stat
import sys

sys.dont_write_bytecode = True
REPO = Path('/home/postedism/Desktop/mini claude code/forge-runtime')
SCOPE = REPO / 'var/lifecycle-rehearsals/lr20260912_a'
OLD_REV = 'c99a285c85d782d5a3d22a84447371db445c69d1'
OLD_SHA = '88288ab0d4717365be52396d3bb5ca8121488f65a540c17c27a02f703064cb3b'
CLEANUP_SHA = 'e7d762c001d1011c03c0569d663f6bd4500990924010d6f26982d521a3d02a7d'
UUID = 'b010dd7440a54c58601967342f397faf686b7e306061af9f49809ee342b0ec8e'
ABSENT = ('evidence/logs-02', 'evidence/host-logs-02-retry', 'runtime/strict-logs-private-02')

def require(ok, message):
    if not ok:
        raise ValueError(message)

def main(revision):
    require(os.getuid() == os.geteuid() == 1000, 'ordinary checkout owner required')
    require(re.fullmatch('[a-f0-9]{40}', revision) and revision != OLD_REV, 'new frozen full revision required')
    spec = importlib.util.spec_from_file_location('logs02_launcher', REPO / 'scripts/faults/sigterm/run.py')
    run = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(run)
    p = run.p
    p.directory(SCOPE / 'runtime', private=True)
    fd = os.open(SCOPE / 'runtime/.sigterm-control.lock', os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC)
    with os.fdopen(fd, 'rb') as lock:
        st = os.fstat(lock.fileno())
        require(stat.S_ISREG(st.st_mode) and st.st_uid == 1000 and st.st_nlink == 1 and stat.S_IMODE(st.st_mode) == 0o600, 'exact private control lock required')
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        base, original = run.inputs('logs', '02', '02')
        require(base == SCOPE and Path(original['test_binary']).parent.name == OLD_REV, 'original prepared c99 input required')
        target = base / 'acceptance-logs-02.json'
        raw = target.read_bytes()
        require(p.digest(target) == OLD_SHA, 'unexecuted preparation changed')
        p.directory(base / 'bin' / revision, private=True)
        build_path = base / 'evidence' / ('build-' + revision) / 'results.json'
        p.directory(build_path.parent, private=True)
        st = build_path.lstat()
        # The reviewed builder stores nonsecret results as0664 inside0700.
        # The private parent excludes other users; preserve original metadata.
        require(stat.S_ISREG(st.st_mode) and st.st_uid == 1000 and st.st_nlink == 1 and not st.st_mode & 0o002 and st.st_size <= 1 << 20, 'owned bounded nonsecret build report required')
        build = json.loads(build_path.read_bytes())
        require(build.get('passed') is True and build.get('source_revision') == revision and build.get('original_binaries_unchanged') is True, 'completed frozen build required')
        require([x['label'] for x in build['commands']] == ['offline', 'race', 'vet', 'runner', 'worker', 'test'] and all(x['exit_code'] == 0 for x in build['commands']), 'all build stages must pass')
        updated = dict(original)
        for label, leaf in (('runner','forge-runner'), ('worker','forge-worker'), ('test','application-faults.test')):
            path = base / 'bin' / revision / leaf
            st = path.lstat()
            require(stat.S_ISREG(st.st_mode) and st.st_uid == 1000 and st.st_nlink == 1 and stat.S_IMODE(st.st_mode) == 0o700, 'private frozen executable required')
            digest = p.digest(path)
            require(build['binaries'][leaf] == {'path':str(path), 'sha256':digest, 'bytes':st.st_size}, 'binary/build identity differs')
            updated[label + '_binary'] = str(path)
            updated[label + '_sha256'] = digest
        binary_fields = {label + suffix for label in ('runner','worker','test') for suffix in ('_binary','_sha256')}
        require({k for k in original if original[k] != updated[k]} <= binary_fields, 'nonbinary authority changed')
        cleanup = base / 'evidence/logs-01-cleanup'
        require(p.digest(cleanup / 'manifest.json') == CLEANUP_SHA, 'historical cleanup manifest changed')
        immutable = {str(cleanup / 'manifest.json'): CLEANUP_SHA}
        for rel, digest in run.private_json(cleanup / 'manifest.json').items():
            path = cleanup / rel
            require(path.is_relative_to(cleanup) and '..' not in Path(rel).parts and not Path(rel).is_absolute(), 'cleanup member path escaped')
            p.directory(path.parent, private=True)
            require(p.digest(path) == digest, 'historical cleanup member changed')
            immutable[str(path)] = digest
        _, cleanup_input = run.inputs('logs-cleanup','02')
        require(cleanup_input == original, 'original cleanup and unexecuted preparation must agree')
        immutable[str(base / 'acceptance-logs-cleanup-01.json')] = p.digest(base / 'acceptance-logs-cleanup-01.json')
        for label in ('runner','worker','test'):
            immutable[original[label+'_binary']] = original[label+'_sha256']
        report = run.private_json(cleanup / 'report.json')
        require(report.get('passed') is True and report.get('released') is True and report.get('snapshot_verified') is True and report.get('journal_uuid') == UUID, 'material cleanup release proof required')
        units = {}
        for host in ('host-logs-02', 'host-logs-cleanup-02'):
            record = run.private_json(base / 'evidence' / host / 'result.json')
            require(record['exit_code'] == 1 and record['unit'].startswith('forge-lifecycle-lr20260912_a-'), 'preserved failed host record required')
            state = run.unit_state(record['unit'])
            require(state.get('LoadState') == 'not-found' and state.get('ActiveState') == 'inactive' and state.get('MainPID') == '0', 'previous exact unit has not exited')
            units[record['unit']] = state
            immutable[str(base / 'evidence' / host / 'result.json')] = p.digest(base / 'evidence' / host / 'result.json')
        def unexecuted():
            require(all(not os.path.lexists(base / rel) for rel in ABSENT), 'logs02 has execution evidence; refuse restamping')
        def settled():
            journal = base / 'runtime/journal.sqlite'
            p.digest(journal)
            with sqlite3.connect(journal.as_uri() + '?mode=ro', uri=True, timeout=3) as db:
                db.execute('PRAGMA query_only=ON')
                db.execute('BEGIN')
                require(db.execute('PRAGMA user_version').fetchone() == (5,), 'journal version changed')
                require(db.execute('SELECT id FROM journal_identity WHERE singleton=1').fetchall() == [(UUID,)], 'journal UUID changed')
                observations = {
                    'workspaces': db.execute('SELECT count(*),sum(released) FROM workspaces').fetchone(),
                    'leases': db.execute('SELECT count(*),sum(released) FROM volume_leases').fetchone(),
                    'operations': db.execute("SELECT count(*),sum(status IN ('succeeded','failed','cancelled')) FROM operations").fetchone(),
                    'logs': db.execute("SELECT count(*),sum(cleanup_state='removed') FROM operation_logs").fetchone(),
                }
                require(observations == {'workspaces':(4,4),'leases':(4,4),'operations':(44,44),'logs':(42,42)}, 'retained journal is not exactly settled')
                return observations
        unexecuted()
        observed = settled()
        destination = base / 'evidence/logs-02-preparation-revision'
        destination.mkdir(mode=0o700)
        archive_parent = os.open(destination.parent, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
        try:
            os.fsync(archive_parent)
        finally:
            os.close(archive_parent)
        p.create_file(destination / 'original-acceptance-logs-02.json', raw)
        prepared = (json.dumps(updated, indent=2, sort_keys=True) + '\n').encode()
        import hashlib
        newsha = hashlib.sha256(prepared).hexdigest()
        p.save(destination / 'intent.json', {'executed':False, 'reason':'Correct final read-only cleanup JSON comparison; actual cleanup and failed exit retained', 'old_sha256':OLD_SHA, 'new_sha256':newsha, 'old_revision':OLD_REV, 'new_revision':revision, 'changed_fields':sorted(k for k in original if original[k] != updated[k]), 'absent_paths':list(ABSENT), 'previous_units':units, 'settled_observations':observed, 'immutable_inputs':immutable, 'script_sha256':p.digest(Path(__file__))})
        temp = base / '.acceptance-logs-02.revision-prepared.json'
        p.create_file(temp, prepared)
        unexecuted()
        settled()
        require(p.digest(target) == OLD_SHA and all(p.digest(Path(path)) == digest for path,digest in immutable.items()), 'authority changed before preparation replacement')
        os.replace(temp, target)
        parent = os.open(base, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
        try:
            os.fsync(parent)
        finally:
            os.close(parent)
        require(run.inputs('logs','02','02')[1] == updated and p.digest(target) == newsha, 'updated preparation verification failed; retain both versions')
        require(all(p.digest(Path(path)) == digest for path,digest in immutable.items()), 'historical authority changed')
        unexecuted()
        p.save(destination / 'result.json', {'prepared':True, 'executed':False, 'new_revision':revision, 'new_sha256':newsha, 'original_copy_sha256':p.digest(destination / 'original-acceptance-logs-02.json'), 'historical_inputs_unchanged':True})
        print(json.dumps({'prepared':True,'executed':False,'evidence':str(destination),'new_revision':revision}))

if __name__ == '__main__':
    main(sys.argv[1])
