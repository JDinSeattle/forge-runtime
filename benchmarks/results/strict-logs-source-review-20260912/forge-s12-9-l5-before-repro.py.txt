#!/usr/bin/env python3
"""Execute the frozen L5 guard expressions, without sockets or fixture state."""
import hashlib
import json
import os
from pathlib import Path
import subprocess

ROOT = Path('/home/postedism/Desktop/mini claude code/forge-runtime')
OUT = Path('/tmp/forge-s12-9-l5-before')
OUT.mkdir(exist_ok=False)
source = (ROOT / 'scripts/faults/application/strict_logs_combined_test.go').read_text()
helpers = (ROOT / 'scripts/faults/application/strict_logs_combined_helpers_test.go').read_text()
assert hashlib.sha256(source.encode()).hexdigest() == 'df7d92567f8c38324a28fc87fd318ab6a4a26e5741e9c390119a1848e11b7bc4'
assert hashlib.sha256(helpers.encode()).hexdigest() == '3f4c7b89f7b1a6470822a16388702ba4e690359d8912d3c163dda176d098562a'
signal_expr = next(line.strip()[3:-2] for line in source.splitlines() if line.strip().startswith('if slNum(signal["exit_code"])'))
ordering_expr = next(line.strip()[3:-2] for line in source.splitlines() if line.strip().startswith('if !exited.After(sent)'))
lease_expr = next(line.strip()[3:-2] for line in source.splitlines() if line.strip().startswith('if claimed.Before(oldUntil)'))
num = helpers[helpers.index('func slNum('):helpers.index('func slIdle(')]
go = '''package main
import("encoding/json";"fmt";"os";"time")
''' + num + '''
func acceptedSignal(signal map[string]any) bool {
 expected := "fixture-binary-sha"
 if ''' + signal_expr + ''' { return false }
 sent,e := time.Parse(time.RFC3339Nano,fmt.Sprint(signal["signal_sent_at"])); if e!=nil{return false}
 exited,e := time.Parse(time.RFC3339Nano,fmt.Sprint(signal["exited_at"])); if e!=nil{return false}
 if ''' + ordering_expr + ''' { return false }; return true
}
func acceptedLease(oldUntil,claimed time.Time) bool {
 if ''' + lease_expr + ''' {return false}; return true
}
func main(){
 baseline := map[string]any{"exit_code":float64(0),"pid":float64(2000),"exe_sha256":"fixture-binary-sha","proc_start_ticks":"456789","signal_sent_at":"2026-09-12T01:00:00Z","exited_at":"2026-09-12T01:00:01Z"}
 results := map[string]any{"positive":acceptedSignal(baseline)}
 clone := func() map[string]any { m:=map[string]any{}; for k,v:=range baseline {m[k]=v}; return m }
 missing:=clone(); delete(missing,"proc_start_ticks"); results["missing_ticks_accepted"]=acceptedSignal(missing)
 pid:=clone(); pid["pid"]=float64(3000); results["replaced_positive_pid_accepted"]=acceptedSignal(pid)
 ticks:=clone(); ticks["proc_start_ticks"]="not-a-number"; results["nonnumeric_ticks_accepted"]=acceptedSignal(ticks)
 fractional:=clone(); fractional["pid"]=float64(2000.5); results["fractional_pid_accepted"]=acceptedSignal(fractional)
 old,_:=time.Parse(time.RFC3339Nano,"2000-01-01T00:00:00Z")
 claim,_:=time.Parse(time.RFC3339Nano,"2000-01-01T00:00:01Z")
 results["unbound_historical_summary_times_accepted"]=acceptedLease(old,claim)
 json.NewEncoder(os.Stdout).Encode(results)
}
'''
(OUT / 'guards.go').write_text(go)
(OUT / 'frozen-test.go.txt').write_text(source)
(OUT / 'frozen-helpers.go.txt').write_text(helpers)
env = {k:v for k,v in os.environ.items() if not k.startswith(('FORGE_', 'OTEL_'))}
tmp = ROOT / 'var/local/s12-independent-go-tmp'
tmp.mkdir(exist_ok=True)
env.update(GOCACHE=str(ROOT/'var/local/go-build-cache'), GOPROXY='off', GOSUMDB='off', GOENV='off', GOTOOLCHAIN='local', TMPDIR='/tmp', GOTMPDIR=str(tmp))
r = subprocess.run(['go','run',str(OUT/'guards.go')], cwd=ROOT, env=env, text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=60)
(OUT / 'stdout.json').write_text(r.stdout)
(OUT / 'stderr.log').write_text(r.stderr)
assert r.returncode == 0, r.stderr
results = json.loads(r.stdout)
assert all(v is True for v in results.values()), results
report = {'scope':'guard-only source-expression extraction; not a full L5 run and no PG, Docker, mount, socket, service, provider or live evidence mutation', 'source_sha256':hashlib.sha256(source.encode()).hexdigest(), 'helpers_sha256':hashlib.sha256(helpers.encode()).hexdigest(), 'exact_extracted_expressions':{'signal':signal_expr, 'exit_order':ordering_expr, 'lease':lease_expr}, 'results':results, 'finding':'P2: evidence guards accept absent/invalid launch identity and unbound historical summary times', 'go_exit':r.returncode}
(OUT/'report.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report,indent=2))
