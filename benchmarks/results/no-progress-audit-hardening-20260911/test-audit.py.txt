import copy
import json
import os
import pathlib
import unittest

import audit


class ProgressAuditRegression(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        root = pathlib.Path(__file__).resolve().parents[2]
        directory = pathlib.Path(os.environ.get("FORGE_PROGRESS_AUDIT_FIXTURES", root / "benchmarks/results/no-progress-20260911/final"))
        cls.directory = directory
        cls.original = json.loads((directory / "TestProgressDurableClosedBatches_read.json").read_text())

    def test_valid_fixture(self):
        self.assertTrue(audit.audit(copy.deepcopy(self.original))["passed"])

    def test_changed_counter_rejected(self):
        data = copy.deepcopy(self.original)
        row = next(r for r in data["run_snapshots"] if (r.get("input_event") or {}).get("progress", {}).get("step_seq") == 1)
        row["body"]["progress"]["repeated_batches"] += 1
        with self.assertRaises(AssertionError):
            audit.audit(data)

    def test_receipt_change_rejected(self):
        data = copy.deepcopy(self.original)
        frame = next(r["input_event"]["progress"] for r in data["run_snapshots"] if (r.get("input_event") or {}).get("progress", {}).get("step_seq") == 1)
        ref = frame["observations"][0]["receipt_ref"]
        op = json.loads(data["artifact_bytes"][ref])
        op["after_hash"] = "f" * 64
        data["artifact_bytes"][ref] = json.dumps(op)
        with self.assertRaises(AssertionError):
            audit.audit(data)

    def test_ledger_argument_change_rejected(self):
        data = copy.deepcopy(self.original)
        item = next(e for e in data["effects"] if e["ordinal"] >= 0)
        item["args_hash"] = "e" * 64
        with self.assertRaises(AssertionError):
            audit.audit(data)

    def test_receipt_and_sql_terminal_status_must_match(self):
        for status in ("unknown", "in_flight", "failed", "cancelled"):
            with self.subTest(status=status):
                data = copy.deepcopy(self.original)
                next(e for e in data["effects"] if e["ordinal"] >= 0)["status"] = status
                with self.assertRaises(AssertionError):
                    audit.audit(data)

    def test_middle_counter_cannot_replace_committed_predecessor(self):
        data = copy.deepcopy(self.original)
        row = next(r for r in data["run_snapshots"] if (r.get("input_event") or {}).get("progress", {}).get("step_seq") == 3)
        self.assertEqual((row["input_state"]["progress"]["repeated_batches"], row["body"]["progress"]["repeated_batches"]), (1, 2))
        row["input_state"]["progress"]["repeated_batches"] = 0
        row["body"]["progress"]["repeated_batches"] = 1
        with self.assertRaises(AssertionError):
            audit.audit(data)

    def test_non_context_transition_cannot_change_progress(self):
        data = copy.deepcopy(self.original)
        rows = sorted(data["run_snapshots"], key=lambda r: r["version"])
        index = next(i for i, r in enumerate(rows) if (r.get("input_event") or {}).get("progress", {}).get("step_seq") == 3)
        # Keep the context arithmetic and every adjacent predecessor equal, but
        # illicitly reset the counter in the preceding non-context transition.
        rows[index - 1]["body"]["progress"]["repeated_batches"] = 0
        rows[index]["input_state"]["progress"]["repeated_batches"] = 0
        rows[index]["body"]["progress"]["repeated_batches"] = 1
        rows[index + 1]["input_state"]["progress"]["repeated_batches"] = 1
        with self.assertRaises(AssertionError):
            audit.audit(data)

    def test_final_current_run_must_match_last_snapshot(self):
        mutations = {"status": "completed", "progress": None, "version": 999}
        for key, value in mutations.items():
            with self.subTest(key=key):
                data = copy.deepcopy(self.original)
                data["run"]["state"][key] = value
                with self.assertRaises(AssertionError):
                    audit.audit(data)

    def test_only_heartbeat_lease_until_may_differ(self):
        data = copy.deepcopy(self.original)
        data["run"]["state"]["lease"]["until"] = "2099-01-01T00:00:00Z"
        self.assertTrue(audit.audit(data)["passed"])
        for key, value in (("owner", "different-owner"), ("epoch", 999)):
            with self.subTest(key=key):
                changed = copy.deepcopy(data)
                changed["run"]["state"]["lease"][key] = value
                with self.assertRaises(AssertionError):
                    audit.audit(changed)

    def test_all_final_original_reports(self):
        for path in sorted(self.directory.glob("TestProgress*.json")):
            with self.subTest(report=path.name):
                self.assertTrue(audit.audit(json.loads(path.read_text()))["passed"])


if __name__ == "__main__":
    unittest.main()
