"""Independent archive-only check. No original evidence/live service/private file access."""
import hashlib,json,pathlib,re,tarfile
P=pathlib.Path('/home/postedism/Desktop/mini claude code/forge-runtime/var/local/feature-worktrees/progress/benchmarks/results/strict-logs-first-failure-20260912')
sha=lambda b:hashlib.sha256(b).hexdigest()
j=lambda n:json.loads((P/n).read_bytes())
a=j('report.json'); idx=j('source-archive-index.json'); scanned=0;env=set(); bad=[]
assert sha((P/'raw-evidence.tar.gz').read_bytes())==a['archive']['sha256']
assert sha((P/'source-archive-index.json').read_bytes())==a['archive']['index_sha256']
assert a['audit_passed'] is True and a['original_overall_passed'] is False
secret={'password','api_key','access_token','refresh_token','bearer_token','signing_key','grant','authorization','database_url','worker_config_sha256'}
def walk(v,p):
 if isinstance(v,dict):
  for k,x in v.items():
   assert not(k.lower() in secret and x),(p,k,'nonempty sensitive field')
   if k=='Env' and isinstance(x,list):
    for e in x:
     if not isinstance(e,str):continue
     name,_,value=e.partition('=');env.add(name)
     assert name in {'GPG_KEY','LANG','PATH','PYTHON_SHA256','PYTHON_VERSION'},(p,'unexpected environment name')
     if name=='GPG_KEY':assert re.fullmatch('[0-9A-F]{40}',value),'public image GPG fingerprint shape'
   walk(x,p+'.'+k)
 elif isinstance(v,list):
  for i,x in enumerate(v):walk(x,p+'['+str(i)+']')
 elif isinstance(v,str) and v[:1] in ('{','['):
  try:x=json.loads(v)
  except (ValueError,RecursionError):return
  walk(x,p+'<json>')
with tarfile.open(P/'raw-evidence.tar.gz') as tf:
 members=tf.getmembers(); assert len(members)==543==len(idx['entries'])
 assert len({x.name for x in members})==len(members)
 for r in idx['entries']:
  m=tf.getmember(r['archive_member']);assert m.isfile() and not m.name.startswith('/') and '..' not in pathlib.PurePosixPath(m.name).parts
  assert m.mtime==0 and m.uid==m.gid==0 and m.mode==0o644
  b=tf.extractfile(m).read();assert len(b)==r['archived_bytes'] and sha(b)==r['archived_sha256']
  if r['transformation']=='byte-identical':assert r['source_sha256']==r['archived_sha256'] and r['source_bytes']==r['archived_bytes']
  else:
   assert m.name=='raw/logs-01/L1/fixture.json';projection=json.loads(b)['_archive_projection']
   assert projection['removed_fields']==['worker_config_sha256'] and projection['source_evidence_sha256']==r['source_sha256']
  if m.name.startswith('raw/') and m.name.endswith(('.json','.meta')):
   try:v=json.loads(b)
   except (ValueError,UnicodeError):continue
   scanned+=1;walk(v,m.name)
 original=json.loads(tf.extractfile('raw/logs-01/manifest.json').read())
 assert len(original)==524
 by={r['archive_member']:r for r in idx['entries']}
 for n,h in original.items(): assert by['raw/logs-01/'+n]['source_sha256']==h
 assert json.loads(tf.extractfile('raw/logs-01/acceptance.json').read())['passed'] is False
 for x in idx['loose_readable_copies']:
  assert sha((P/x['path']).read_bytes())==x['sha256']==by[x['archive_member']]['archived_sha256']
result={'passed':True,'scope':'archive-only readback, original-manifest to source-index coverage, explicit projection and structural credential-field scan; not a runtime suite pass','archive_members':len(members),'original_manifest_members':len(original),'archive_sha256':a['archive']['sha256'],'index_sha256':a['archive']['index_sha256'],'report_sha256':sha((P/'report.json').read_bytes()),'json_files_scanned':scanned,'nonempty_sensitive_fields':[],'container_environment_names':sorted(env),'container_GPG_KEY_class':'public 40-hex image signing fingerprint, not private material','original_overall_passed':False}
print(json.dumps(result,indent=2,sort_keys=True))
