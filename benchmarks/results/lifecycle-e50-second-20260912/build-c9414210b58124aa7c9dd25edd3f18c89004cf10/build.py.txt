import hashlib
import importlib.util
import json
import os
from pathlib import Path
import re
import stat
import subprocess
import sys
import time

repo = Path('/home/postedism/Desktop/mini claude code/forge-runtime')
scope = repo / 'var/lifecycle-rehearsals/lr20260912_a'
go = Path('/home/postedism/.local/share/mise/installs/go/1.26.8/bin/go')
revision = sys.argv[1]
sys.dont_write_bytecode = True
assert re.fullmatch('[a-f0-9]{40}', revision)
assert Path.cwd() == repo and os.getuid() == os.geteuid() == 1000
def git(*args):
    return subprocess.check_output(['git', *args], cwd=repo).decode().strip()
assert git('rev-parse', 'HEAD') == revision
assert not git('status', '--porcelain')
def sha(path):
    st = path.lstat()
    assert stat.S_ISREG(st.st_mode) and st.st_nlink == 1
    with path.open('rb') as f:
        return hashlib.file_digest(f, 'sha256').hexdigest()
def snapshot():
    paths = [Path(p) for p in git('ls-files', '-z').split('\0') if p and not p.startswith('benchmarks/results/')]
    result = {str(p): sha(repo/p) for p in paths}
    for path, digest in result.items():
        data = subprocess.check_output(['git','show',revision+':'+path],cwd=repo)
        assert hashlib.sha256(data).hexdigest() == digest, path
    return result
before = snapshot()
def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module
launcher = load('frozen_lifecycle_launch',repo/'scripts/faults/sigterm/run.py')
bounded = load('bounded_lifecycle_build',repo/'tests/review/run_api_lifecycle.py')
validated_scope, original = launcher.inputs()
assert validated_scope == scope
for path in (scope/'bin',scope/'evidence'):
    launcher.p.directory(path,private=True)
old_names = {'forge-runner','forge-worker','application-faults.test'}
def original_binaries():
    assert {p.name for p in (scope/'bin').iterdir() if not p.is_dir()} == old_names
    return {name:sha(scope/'bin'/name) for name in old_names}
old = original_binaries()
for label, name in [('runner','forge-runner'),('worker','forge-worker'),('test','application-faults.test')]:
    assert original[label+'_sha256'] == old[name]
out = scope/'evidence'/('build-'+revision)
out.mkdir(mode=0o700)
binary_dir = scope/'bin'/revision
binary_dir.mkdir(mode=0o700)
launcher.p.directory(binary_dir,private=True)
launcher.p.directory(out,private=True)
env = {'PATH':str(go.parent)+':/usr/bin:/bin','HOME':str(scope),'LC_ALL':'C','TMPDIR':'/tmp',
       'GOPROXY':'off','GOSUMDB':'off','GOENV':'off','GOTOOLCHAIN':'local','CGO_ENABLED':'1',
       'GOCACHE':str(repo/'var/local/go-build-cache'),'GOTMPDIR':str(repo/'var/local/integration-go-tmp'),
       'GOPATH':'/home/postedism/go','GOMODCACHE':'/home/postedism/go/pkg/mod'}
def save(name,data):
    with (out/name).open('x') as f:json.dump(data,f,indent=2);f.write('\n')
save('inputs-before.json',before)
save('intent.json',{'source_revision':revision,'old_binaries':old,'environment':env,'build_script_sha256':sha(Path(__file__)),
                   'go_sha256':sha(go),'go_version':subprocess.check_output([str(go),'version'],env=env).decode().strip(),
                   'scope':'new versioned binaries and offline integration only; no service, DB, Docker or model calls'})
(out/'build.py.txt').write_bytes(Path(__file__).read_bytes())
results=[]
packages=['./internal/provider','./internal/configuration','./internal/application','./internal/sandbox','./scripts/faults/application']
commands=[('offline',[str(go),'test','-count=1','-json',*packages]),('race',[str(go),'test','-race','-count=1','-json',*packages]),
          ('vet',[str(go),'vet',*packages]),
          ('runner',[str(go),'build','-buildvcs=true','-o',str(binary_dir/'forge-runner'),'./cmd/forge-runner']),
          ('worker',[str(go),'build','-buildvcs=true','-o',str(binary_dir/'forge-worker'),'./cmd/forge-worker']),
          ('test',[str(go),'test','-c','-o',str(binary_dir/'application-faults.test'),'./scripts/faults/application'])]
for label,argv in commands:
    started=time.monotonic()
    with (out/(label+'.log')).open('xb') as log:
        code=bounded.execute_bounded(argv,repo,env,log,240)
    results.append({'label':label,'argv':argv,'exit_code':code,'elapsed_seconds':time.monotonic()-started})
    print(json.dumps(results[-1]),flush=True)
    if code:
        save('failed-results.json',results)
        sys.exit(code)
binaries={}
for path in binary_dir.iterdir():
    path.chmod(0o700)
    metadata=subprocess.check_output([str(go),'version','-m',str(path)],env=env,cwd=repo).decode()
    (out/(path.name+'.buildinfo.txt')).write_text(metadata)
    if path.name!='application-faults.test':
        assert 'vcs.revision='+revision in metadata and 'vcs.modified=false' in metadata
    binaries[path.name]={'sha256':sha(path),'bytes':path.stat().st_size,'path':str(path)}
after=snapshot()
save('inputs-after.json',after)
assert before==after and not git('status','--porcelain') and git('rev-parse','HEAD')==revision
assert old==original_binaries()
save('results.json',{'passed':True,'source_revision':revision,'unchanged_inputs':len(before),'original_binaries_unchanged':True,'commands':results,'binaries':binaries,'scope':'offline integration and build only; actual acceptance not executed'})
print(json.dumps({'passed':True,'evidence':str(out),'binary_dir':str(binary_dir)}),flush=True)
